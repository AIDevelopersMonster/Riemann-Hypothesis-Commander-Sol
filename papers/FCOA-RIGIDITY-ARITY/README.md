\[
q=2:\quad k_{exact}=3,
\qquad
q\ge3:\quad k_{exact}=4.
\]

\[
Q(x,y,z)\iff x\star y=y\star z.

\[
E(x,y,u,v)\iff x\star y=u\star v.
\]
